```javascript
app.get('/api/announcements', (req, res) => {
    const posts = getAllPosts();
    
    // Sort: pinned posts first (by pinnedAt desc), then regular posts by createdAt desc
    const sortedPosts = posts.sort((a, b) => {
        // First, sort by pinned status
        if (a.isPinned && !b.isPinned) return -1;
        if (!a.isPinned && b.isPinned) return 1;
        
        // If both pinned, sort by pinnedAt (most recently pinned first)
        if (a.isPinned && b.isPinned) {
            return new Date(b.pinnedAt) - new Date(a.pinnedAt);
        }
        
        // If both not pinned, sort by createdAt (newest first)
        return new Date(b.createdAt) - new Date(a.createdAt);
    });
    
    res.json(sortedPosts);
});
```

### 3. Attachments Not Displaying
**Problem:** Pictures and polls are not showing up in posts.

**Fix Required:**

#### Frontend Post Rendering Fix:
```javascript
function createPostElement(post) {
    const postDiv = document.createElement('div');
    postDiv.className = 'post-card';
    
    let attachmentsHTML = '';
    
    // Handle Images
    if (post.attachments && post.attachments.images && post.attachments.images.length > 0) {
        attachmentsHTML += '<div class="post-images">';
        post.attachments.images.forEach(imageUrl => {
            attachmentsHTML += `<img src="${imageUrl}" class="post-image" alt="Attachment">`;
        });
        attachmentsHTML += '</div>';
    }
    
    // Handle Videos
    if (post.attachments && post.attachments.videos && post.attachments.videos.length > 0) {
        attachmentsHTML += '<div class="post-videos">';
        post.attachments.videos.forEach(videoUrl => {
            attachmentsHTML += `<video controls class="post-video"><source src="${videoUrl}"></video>`;
        });
        attachmentsHTML += '</div>';
    }
    
    // Handle Polls
    let pollHTML = '';
    if (post.poll) {
        pollHTML = `
            <div class="poll-container">
                <h4 class="poll-question">${post.poll.question}</h4>
                <div class="poll-options">
                    ${post.poll.options.map((option, index) => {
                        const votes = post.poll.votes[`option${index + 1}`] || [];
                        const percentage = post.poll.totalVotes ? (votes.length / post.poll.totalVotes * 100).toFixed(1) : 0;
                        return `
                            <div class="poll-option" onclick="voteInPoll('${post.id}', ${index})">
                                <span class="poll-option-text">${option}</span>
                                <div class="poll-bar">
                                    <div class="poll-bar-fill" style="width: ${percentage}%"></div>
                                </div>
                                <span class="poll-percentage">${percentage}%</span>
                            </div>
                        `;
                    }).join('')}
                </div>
                <div class="poll-info">${post.poll.totalVotes || 0} votes</div>
            </div>
        `;
    }
    
    postDiv.innerHTML = `
        <div class="post-content">${post.content}</div>
        ${attachmentsHTML}
        ${pollHTML}
        <div class="post-reactions">
            <!-- reaction buttons -->
        </div>
    `;
    
    return postDiv;
}
```

#### Backend File Upload Fix:
```javascript
// Make sure file upload is properly configured
const multer = require('multer');
const path = require('path');

const storage = multer.diskStorage({
    destination: (req, file, cb) => {
        cb(null, 'uploads/');
    },
    filename: (req, file, cb) => {
        cb(null, Date.now() + path.extname(file.originalname));
    }
});

const upload = multer({ storage: storage });

// File upload endpoint
app.post('/api/upload', upload.single('file'), (req, res) => {
    if (!req.file) {
        return res.status(400).json({ error: 'No file uploaded' });
    }
    
    const fileUrl = `/uploads/${req.file.filename}`;
    res.json({ url: fileUrl });
});

// Serve uploaded files
app.use('/uploads', express.static('uploads'));
```
