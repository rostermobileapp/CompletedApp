**Fix Required:**

#### Backend: Store Reactions Correctly
```javascript
// Make sure your post schema stores reactions like this:

    reactions: {
        "👍": ["user1", "user2", "user3"],  // Array of user IDs who reacted

// API endpoint to add/remove reactions
app.post('/api/announcements/:postId/react', (req, res) => {
    const { postId } = req.params;
    const { emoji, action } = req.body; // action: 'add' or 'remove'
    const userId = req.user.id; // Get from session/auth
    
    const post = getPostById(postId);
    if (!post.reactions) post.reactions = {};
    if (!post.reactions[emoji]) post.reactions[emoji] = [];
    
    if (action === 'add') {
        // Add user to reaction if not already there
        if (!post.reactions[emoji].includes(userId)) {
            post.reactions[emoji].push(userId);
        }
    } else if (action === 'remove') {
        // Remove user from reaction
        post.reactions[emoji] = post.reactions[emoji].filter(id => id !== userId);
        // Clean up empty reaction arrays
        if (post.reactions[emoji].length === 0) {
            delete post.reactions[emoji];
        }
    }
    
    // Save updated post to database
    updatePost(postId, { reactions: post.reactions });
    
    // Return updated reactions
    res.json({ reactions: post.reactions });
});
```

// Function to toggle reactions
async function toggleReaction(postId, emoji) {
    const userId = getCurrentUserId(); // You need to implement this
    const post = getPostFromDOM(postId); // Get current post data
    
    // Check if user already reacted with this emoji
    const hasReacted = post.reactions && post.reactions[emoji] && post.reactions[emoji].includes(userId);
    const action = hasReacted ? 'remove' : 'add';
    
    try {
        const response = await fetch(`/api/announcements/${postId}/react`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ emoji, action })
        });
        
        if (response.ok) {
            const data = await response.json();
            // Update the post display with new reaction counts
            updatePostReactions(postId, data.reactions);
        }
    } catch (error) {
        console.error('Error toggling reaction:', error);
    }
}

// Function to update reaction display
function updatePostReactions(postId, reactions) {
    const postElement = document.querySelector(`[data-post-id="${postId}"]`);
    const reactionsDisplay = postElement.querySelector('.post-reactions-display');
    
    let reactionsHTML = '';
    if (reactions && Object.keys(reactions).length > 0) {
        reactionsHTML = '<div class="post-reactions-display">';
        
        for (const [emoji, users] of Object.entries(reactions)) {
            const count = users.length;
            if (count > 0) {
                reactionsHTML += `
                    <span class="reaction-item">
                        ${emoji} ${count}
                    </span>
                `;
            }
        }
        reactionsHTML += '</div>';
    }
    
    if (reactionsDisplay) {
        reactionsDisplay.outerHTML = reactionsHTML;
    } else if (reactionsHTML) {
        // Insert reactions display if it doesn't exist
        const contentDiv = postElement.querySelector('.post-content');
        contentDiv.insertAdjacentHTML('afterend', reactionsHTML);
    }
}

function calculateTotalVotes(votes) {
    let total = 0;
    Object.values(votes).forEach(voteArray => {
        if (Array.isArray(voteArray)) {
            total += voteArray.length;
        }
    });

}
